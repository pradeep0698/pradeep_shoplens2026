# CookShop — Flutter Android App Implementation Guide

> **Zero backend changes required.** This app connects entirely to the existing deployed Google Cloud infrastructure. Every Cloud Run service, Firebase project, Firestore collection, and GCP resource already exists and is shared as-is with the web app.

```dart
  SessionApi(this._dio);

  // POST /session/{sessionId}/products — same endpoint the web app calls after matching
  Future<void> saveProducts(String sessionId, List<Product> products) async {
    await _dio.post(
      '/session/$sessionId/products',
      data: {'products': products.map(Product.toFirestore).toList()},
    );
  }

  // GET /session/{sessionId} — hydrates session on app launch
  Future<List<Product>> loadSession(String sessionId) async {
    final response = await _dio.get('/session/$sessionId');
    final data     = response.data as Map<String, dynamic>;
    final list     = data['products'] as List<dynamic>? ?? [];
    return list
        .map((e) => Product.fromFirestore(e as Map<String, dynamic>))
        .toList();
  }

  // DELETE /session/{sessionId} — same as the web app Reset button
  Future<void> clearSession(String sessionId) async {
    await _dio.delete('/session/$sessionId');
  }
}
```

```dart
import 'package:cloud_firestore/cloud_firestore.dart';
import '../../models/user_profile.dart';
import '../../models/product.dart';
import '../../../core/constants/firestore_constants.dart';

class FirestoreSource {
  final _db = FirebaseFirestore.instance;

  // Watches the existing UserProfiles/{uid} document
  // Same collection and field names the web app writes
  Stream<UserProfile> watchProfile(String uid) =>
      _db
          .collection(FirestoreConstants.userProfiles)
          .doc(uid)
          .snapshots()
          .map((snap) => snap.exists
              ? UserProfile.fromFirestore(snap.data()!)
              : const UserProfile());

  Future<void> saveProfile(String uid, UserProfile profile) =>
      _db
          .collection(FirestoreConstants.userProfiles)
          .doc(uid)
          .set(UserProfile.toFirestore(profile), SetOptions(merge: true));

  // Watches the existing LiveShoppingSessions/{sessionId} document
  // sessionId must be "cookshop-user-{uid}" — must match State Manager derivation exactly
  Stream<List<Product>> watchShoppingList(String sessionId) =>
      _db
          .collection(FirestoreConstants.shoppingSessions)
          .doc(sessionId)
          .snapshots()
          .map((snap) {
            if (!snap.exists) return <Product>[];
            final list = snap.data()!['products'] as List<dynamic>? ?? [];
            return list
                .map((e) => Product.fromFirestore(e as Map<String, dynamic>))
                .toList();
          });
}

final firestoreSourceProvider = Provider((_) => FirestoreSource());
```

## Step 8 — Product Ranker

### `lib/core/utils/product_ranker.dart`

```dart
// Replicates the preference ranking logic from frontend/src/app/page.tsx
// Products whose names contain any preferenceTerms float to the top
List<Product> rankProducts(
  List<Product> products,
  List<String>  preferenceTerms,
) {
  if (preferenceTerms.isEmpty) return products;

  final preferred = <Product>[];
  final rest      = <Product>[];

  for (final product in products) {
    final name = product.name.toLowerCase();
    final isPreferred = preferenceTerms.any(
      (term) => name.contains(term.toLowerCase()),
    );
    isPreferred ? preferred.add(product) : rest.add(product);
  }

  return [...preferred, ...rest];
}
```

## Step 10 — Analyze Image Use Case

### `lib/domain/usecases/analyze_image_usecase.dart`

```dart
import 'dart:typed_data';

// Orchestrates the same 3-step sequential pipeline the web app runs client-side:
//   Step 1 → POST /analyze  (existing AI Analyzer Cloud Run)
//   Step 2 → POST /match    (existing Product Matcher Cloud Run)
//   Step 3 → POST /session  (existing State Manager Cloud Run) → writes Firestore
// Firestore onSnapshot (via watchShoppingList) automatically pushes the update to the UI.

enum PipelineStep { analyzing, matching, saving, done }

class AnalyzeImageUseCase {
  final AnalyzerApi _analyzer;
  final MatcherApi  _matcher;
  final SessionApi  _session;

  AnalyzeImageUseCase(this._analyzer, this._matcher, this._session);

  Stream<PipelineStep> execute({
    required Uint8List    imageBytes,
    required String       mimeType,
    required String       sessionId,
    required List<String> ignoreTerms,
    required List<String> preferenceTerms,
  }) async* {
    // Step 1 — existing AI Analyzer Cloud Run service
    yield PipelineStep.analyzing;
    final analyzeResponse = await _analyzer.analyze(AnalyzeRequest(
      imageData:     encodeImageToBase64(imageBytes),
      imageMimeType: mimeType,
      ignoreTerms:   ignoreTerms,
      transcript:    '',
    ));

    // Step 2 — existing Product Matcher Cloud Run service
    yield PipelineStep.matching;
    final matchResponse = await _matcher.match(MatchRequest(
      items:       analyzeResponse.items,
      ignoreTerms: ignoreTerms,
    ));

    final ranked = rankProducts(matchResponse.matchedProducts, preferenceTerms);

    // Step 3 — existing State Manager Cloud Run service
    // State Manager writes to Firestore → triggers watchShoppingList stream → UI rebuilds
    yield PipelineStep.saving;
    await _session.saveProducts(sessionId, ranked);

    yield PipelineStep.done;
  }
}

final analyzeImageUseCaseProvider = Provider((ref) => AnalyzeImageUseCase(
      ref.read(analyzerApiProvider),
      ref.read(matcherApiProvider),
      ref.read(sessionApiProvider),
    ));
```

### `lib/presentation/providers/pipeline_provider.dart`

```dart
import 'package:freezed_annotation/freezed_annotation.dart';

part 'pipeline_provider.freezed.dart';

enum PipelineStatus { idle, analyzing, matching, saving, success, error }

@freezed
class PipelineState with _$PipelineState {
  const factory PipelineState({
    @Default(PipelineStatus.idle) PipelineStatus status,
    String? errorMessage,
  }) = _PipelineState;
}

class PipelineNotifier extends AutoDisposeNotifier<PipelineState> {
  @override PipelineState build() => const PipelineState();

  Future<void> analyzeImage(Uint8List bytes, String mimeType) async {
    final user    = ref.read(authStateProvider).value;
    final profile = ref.read(profileProvider).value ?? const UserProfile();
    if (user == null) return;

    final sessionId = getSessionId(user.uid);
    state = const PipelineState(status: PipelineStatus.analyzing);

    try {
      await for (final step in ref.read(analyzeImageUseCaseProvider).execute(
        imageBytes:      bytes,
        mimeType:        mimeType,
        sessionId:       sessionId,
        ignoreTerms:     profile.ignoreTerms,
        preferenceTerms: profile.preferenceTerms,
      )) {
        state = PipelineState(
          status: switch (step) {
            PipelineStep.analyzing => PipelineStatus.analyzing,
            PipelineStep.matching  => PipelineStatus.matching,
            PipelineStep.saving    => PipelineStatus.saving,
            PipelineStep.done      => PipelineStatus.success,
          },
        );
      }
    } catch (e) {
      state = PipelineState(
        status:       PipelineStatus.error,
        errorMessage: e.toString(),
      );
    }
  }

  Future<void> clearSession() async {
    final user = ref.read(authStateProvider).value;
    if (user == null) return;
    await ref.read(sessionRepositoryProvider).clear(getSessionId(user.uid));
  }
}

final pipelineProvider =
    AutoDisposeNotifierProvider<PipelineNotifier, PipelineState>(
        PipelineNotifier.new);
```
